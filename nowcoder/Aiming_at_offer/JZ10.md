# JZ10.斐波那契数列

```cpp
class Solution {
public:
    std::vector<int> fib = std::vector<int>(50, 0);
    int Fibonacci(int n) {
        if(fib[n] > 0) return fib[n];
        if(n == 1 || n == 2){
            fib[1] = fib[2] = 1;
            return 1;
        }
        fib[n] = Fibonacci(n - 1) + Fibonacci(n - 2);
        return fib[n];
    }
};
```
