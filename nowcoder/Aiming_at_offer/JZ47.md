# JZ47.礼物的最大价值

```cpp
class Solution {
public:
    int maxValue(vector<vector<int> >& grid) {
        // m 行 n 列
        int m = grid.size();
        int n = grid[0].size();
        // 转移最右侧一列
        for(int i = m - 1; i > 0; i --) grid[i - 1][n - 1] += grid[i][n - 1];
        // 转移最下侧一行
        for(int i = n - 1; i > 0; i --) grid[m - 1][i - 1] += grid[m - 1][i]; 
        for(int i = m - 2; i >= 0; i --){
            for(int j = n - 2; j >=0 ;j --){
                grid[i][j] += std::max(grid[i][j + 1], grid[i + 1][j]);
            }
        }
        for(int i = 0; i < m; i ++){
            for(int j = 0; j < n; j++){
                std::cout << grid[i][j] << ' ';
            }
            std::cout << '\n';
        }
        return grid[0][0];
    }
};
```
