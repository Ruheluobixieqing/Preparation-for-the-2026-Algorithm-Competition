# JZ85.连续子数组的最大和（二）

```cpp
class Solution {
public:
    vector<int> FindGreatestSumOfSubArray(vector<int>& array) {
        std::vector<int> res;
        std::vector<int> dp(array.size() ,0);
        int maxSum = array[0];
        dp[0] = array[0];
        int left = 0, right = 0;
        int resLeft = 0, resRight = 0;
        for(int i = 1; i < array.size(); i++){
            right ++;
            dp[i] = std::max(dp[i - 1] + array[i], array[i]);
            if(dp[i - 1] + array[i] < array[i]) left = right;
            if(dp[i] > maxSum || (dp[i] == maxSum && (right - left + 1) > (resRight - resLeft + 1))){
                maxSum = dp[i];
                resLeft = left, resRight = right;
            }
        }
        for(int i = resLeft; i <= resRight; i++){
            res.push_back(array[i]);
        }
        return res;
    }
};
```
