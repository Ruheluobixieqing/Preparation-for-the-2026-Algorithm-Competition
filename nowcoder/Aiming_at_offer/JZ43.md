# JZ43.整数中 1 出现的次数（从 1 到 n 整数中 1 出现的次数）

```cpp
class Solution {
public:
    int NumberOf1Between1AndN_Solution(int n) {
        int base = 1;
        int ans = 0;
        while(base <= n){
            int cur = n / base % 10;
            int a = n / base / 10;
            int b = n % base;
            if(cur == 1) ans += b + 1 + a * base;
            else if( cur == 0) ans += a * base;
            else ans += (a + 1) * base;
            base *= 10;
            std::cout << ans << '\n';
        }
        return ans;
    }
};
```
