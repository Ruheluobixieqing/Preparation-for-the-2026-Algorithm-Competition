# JZ44.数字序列中某一位的数组

```cpp
class Solution {
public:
    int findNthDigit(int n) {
        int digit = 1;
        long long start = 1;
        long long sum = 9;
        // 锁定 n 的位数
        while(n > sum){
            n -= sum;
            start *= 10;
            digit ++;
            sum = 9 * start * digit;
        }
        int num = start + (n - 1) / digit;
        int index = (n - 1) % digit;
        int ans = 0;
        while(index < digit){
            ans = num % 10;
            num /= 10;
            index ++;
        }
        return ans;
    }
};
```
