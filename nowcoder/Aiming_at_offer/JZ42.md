# JZ42.连续子数组的最大和

```cpp
class Solution {
public:
    int FindGreatestSumOfSubArray(vector<int>& array) {
        int n = array.size();
        if(n == 0) return 0;
        int tmp = array[0];
        int ans = tmp;
        for(int i = 1; i < n; i ++){
            tmp = std::max(tmp + array[i], array[i]);
            ans = std::max(ans, tmp);
        }
        return ans;
    }
};
```
