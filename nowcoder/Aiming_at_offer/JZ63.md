# JZ63.买股票的最好时机（一）

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int ans = 0;
        if(prices.size() == 0) return ans;
        int minPrice = prices[0];
        std::cout << minPrice << '\n';
        for(int i = 1; i < prices.size(); i++){
            ans = std::max(ans, prices[i] - minPrice);
            minPrice = std::min(minPrice, prices[i]);
        }
        return ans;
    }
};
```
