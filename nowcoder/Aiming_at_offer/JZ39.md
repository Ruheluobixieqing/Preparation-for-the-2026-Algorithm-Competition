# JZ39.数组中出现次数超过一半的数字

```cpp
class Solution {
public:
    int MoreThanHalfNum_Solution(vector<int>& numbers) {
        std::map<int, int> mp;
        int n = numbers.size();
        for(int i = 0; i < n; i++){
            mp[numbers[i]]++;
            if(mp[numbers[i]] * 2 > n) return numbers[i];
        }
        return 0;
    }
};
```
