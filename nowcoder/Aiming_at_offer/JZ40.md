# JZ40.最小的 k 个数

```cpp
class Solution {
public:
    vector<int> GetLeastNumbers_Solution(vector<int>& input, int k) {
        std::vector<int> ans;
        std::priority_queue<int> pq;
        for(const auto e : input){
            if(pq.size() < k) {
                pq.push(e);
                continue;
            }
            if(!pq.empty() && pq.top() > e){
                pq.pop();
                pq.push(e);
            }
        }
        while(!pq.empty()){
            std::cout<<pq.top()<< ' ';
            ans.push_back(pq.top());
            pq.pop();
        }
        return ans;
    }
};
```
