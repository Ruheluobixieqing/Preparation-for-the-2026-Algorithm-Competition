# JZ41.数据流的中位数

```cpp
class Solution {
public:
    std::priority_queue<int> leftArr;
    std::priority_queue<int, std::vector<int>, std::greater<int>> rightArr;

    void Insert(int num) {
        if(leftArr.size() <= rightArr.size()) leftArr.push(num);
        else rightArr.push(num);
        std::cout<<leftArr.size() << ' ' << rightArr.size() << '\n';
        if(leftArr.size() > 0 && rightArr.size() > 0 && leftArr.top() > rightArr.top()){
            int tmp = leftArr.top();
            leftArr.pop();
            leftArr.push(rightArr.top());
            rightArr.pop();
            rightArr.push(tmp);
        }
    }

    double GetMedian() { 
        if(leftArr.size() == rightArr.size()) return (leftArr.top() + rightArr.top()) / 2.0;
        else if(leftArr.size() > rightArr.size()) return leftArr.top() * 1.0;
        else return rightArr.top() * 1.0;
    }
};
```
