# JZ38.字符串的排列

```cpp
class Solution {
public:
    void recursion(std::vector<std::string> &res, std::string &str, std::string &temp, std::vector<int> &vis){
        if(temp.length() == str.length()){
            res.push_back(temp);
            return;
        }
        for(int i = 0; i < str.length(); i ++){
            if(vis[i] || ((i > 0 && str[i] == str[i - 1] && vis[i - 1]))) continue;
            vis[i] = 1;
            temp.push_back(str[i]);
            recursion(res, str, temp, vis);
            vis[i] = 0;
            temp.pop_back();
        }
    }
    
    vector<string> Permutation(string str) {
        std::sort(str.begin(), str.end());
        std::vector<std::string> res;
        std::string temp = "";
        std::vector<int> vis(str.length(), 0);
        recursion(res, str, temp, vis);
        return res;
    }
};
```
