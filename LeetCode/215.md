# 215.数组中第 K 个最大元素

```cpp
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        std::priority_queue<int, std::vector<int>, std::greater<int>> pq;
        for(auto e : nums) {
            if(pq.size() < k) pq.push(e);
            else {
                if(e > pq.top()) {
                    pq.pop();
                    pq.push(e);
                }
            }
        }
        return pq.top();
    }
};
```
