# 53.最大子数组和

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int ans = nums[0];
        int sum = ans;
        for(int right = 1; right < nums.size(); right++){
            sum = std::max(sum + nums[right], nums[right]);
            ans = std::max(ans, sum);
        }
        return ans;
    }
};
```
