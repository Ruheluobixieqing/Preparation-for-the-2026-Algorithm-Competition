# 460.LFU 缓存

```cpp
struct Node {
    int cnt;
    int time;
    int key;
    int value;
    
    Node(int _cnt, int _time, int _key, int _value): cnt(_cnt), time(_time), key(_key), value(_value) {}

    bool operator< (const Node& rhs) const{ 
        return this -> cnt == rhs.cnt ? this -> time < rhs.time : this -> cnt < rhs.cnt;
    }
};

class LFUCache {
private:
    int time;
    int capacity;
    std::set<Node> S;
    std::unordered_map<int, Node> cache;

public:
    LFUCache(int capacity) {
        this -> capacity = capacity;
        time = 0;
        S.clear();
        cache.clear();
    }
    
    int get(int key) {
        time ++;
        auto it = cache.find(key);
        if(it == cache.end()) return -1;
        Node node = it -> second;
        S.erase(node);
        node.time = time;
        node.cnt ++;
        S.insert(node);
        it -> second = node;
        return node.value;
    }
    
    void put(int key, int value) {
        time ++;
        auto it = cache.find(key);
        if(it == cache.end()){
            if(cache.size() == capacity){
                cache.erase(S.begin() -> key);
                S.erase(S.begin());
            }
            Node node = Node(1, time, key, value);
            cache.insert(std::make_pair(key, node));
            S.insert(node);
        }
        else {
            Node node = it -> second;
            S.erase(node);
            node.cnt ++;
            node.time = time;
            node.value = value;
            S.insert(node);
            it -> second = node;
        }
    }
};
```
