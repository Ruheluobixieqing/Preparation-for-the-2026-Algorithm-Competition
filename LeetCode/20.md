# 20.有效的括号

```cpp
class Solution {
public:
    std::stack<char> st;
    bool isValid(string s) {
        if(s.size() & 1) return false;
        for(int i = 0; i < s.size(); i++){
            if(s[i] == '(' || s[i] == '[' || s[i] == '{') st.push(s[i]);
            else {
                if(st.empty()) return false;
                if(s[i] == ')' && st.top() != '(') return false;
                if(s[i] == ']' && st.top() != '[') return false;
                if(s[i] == '}' && st.top() != '{') return false;
                st.pop();
            }
        }
        return st.empty();
    }
};
```
