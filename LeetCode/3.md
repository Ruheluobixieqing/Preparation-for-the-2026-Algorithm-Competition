# 3.无重复字符的最长子串

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        if(s.size() == 0) return 0;
        std::map<char, int> mp;
        int left = 0, ans = 1;
        mp[s[0]] = 0;
        for(int i = 1; i < s.size(); i++){
            if(mp.find(s[i]) != mp.end() && mp[s[i]] >= left){
                left = mp[s[i]] + 1;
                mp[s[i]] = i;
            }
            else {
                ans = std::max(ans, i - left + 1);
                mp[s[i]] = i;
            }
        }
        return ans;
    }
};
```
