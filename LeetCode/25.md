# 25.K 个一组翻转链表

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    std::stack<ListNode*> st;
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode* ans = new ListNode();
        ans -> next = head;
        ListNode* pretail = ans;
        ListNode* p = head;
        ListNode* start = head;
        ListNode* end;
        while(p != nullptr && st.size() < k){
            st.push(p);
            end = p;
            p = p -> next;
        }
        if(p == nullptr && st.size() != k) return head;
        ans -> next = end;
        while(st.size() > 1){
            ListNode* ptr = st.top();
            st.pop();
            ptr -> next = st.top();
        }
        start = st.top();
        st.pop();
        start -> next = p;
        pretail -> next = end;
        pretail = start;
        start = p;
        while(p != nullptr && st.size() < k){
            st.push(p);
            end = p;
            p = p -> next;
            if(st.size() == k){
                while(st.size() > 1){
                    ListNode* ptr = st.top();
                    st.pop();
                    ptr -> next = st.top();
                }
                start = st.top();
                st.pop();
                pretail -> next = end;
                start -> next = p;
                pretail = start;
            }
        }
        return ans -> next;
    }
};
```
