# 23.合并 K 个升序链表

```cpp
class Solution {
private:
    struct Status {
        int val;
        ListNode* ptr;

        Status(int _val, ListNode* _ptr): val(_val), ptr(_ptr) {}

        bool operator< (const Status &rhs) const {
            return val > rhs.val;
        }
    };

    std::priority_queue<Status> pq;

public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        for(auto node: lists) {
            if(node) {
                Status this_node = Status(node -> val, node);
                pq.push(this_node);
            }
        }
        ListNode* head = new ListNode();
        ListNode* tail = head;
        while(!pq.empty()){
            auto it = pq.top();
            pq.pop();
            tail -> next = it.ptr;
            tail = tail -> next;
            if(it.ptr -> next) {
                Status next_node = Status(it.ptr -> next -> val, it.ptr -> next);
                pq.push(next_node);
            }
        }
        return head -> next;

    }
};
```
